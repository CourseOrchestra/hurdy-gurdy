package com.example.controller;

import com.example.dto.UploadMeta;
import jakarta.servlet.http.HttpServletResponse;
import java.lang.String;
import java.util.List;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestPart;
import org.springframework.web.multipart.MultipartFile;

public interface Controller {
  @PostMapping(
      value = "/upload",
      produces = "application/json",
      consumes = "multipart/form-data"
  )
  UploadMeta upload(@RequestPart(name = "one") MultipartFile one,
      @RequestPart(name = "files") List<MultipartFile> files,
      @RequestPart(name = "attachments") List<MultipartFile> attachments,
      @RequestPart(name = "tags") List<String> tags, @RequestPart(name = "meta") UploadMeta meta,
      HttpServletResponse response);
}
