package com.example.controller

import com.example.dto.Sample
import com.example.dto.SampleUpdate
import kotlin.Boolean
import kotlin.Int
import kotlin.String
import kotlin.collections.List
import org.springframework.web.bind.`annotation`.GetMapping
import org.springframework.web.bind.`annotation`.PathVariable
import org.springframework.web.bind.`annotation`.PostMapping
import org.springframework.web.bind.`annotation`.PutMapping
import org.springframework.web.bind.`annotation`.RequestBody
import org.springframework.web.bind.`annotation`.RequestParam

public interface Controller {
  @GetMapping(
    value = ["/api/v1/sample/{sample_id}"],
    produces = ["application/json"],
  )
  public fun getSample(@PathVariable(name = "sample_id") sampleId: String?, @RequestParam(required = false, name = "page") page: Int?): Sample?

  @PostMapping(
    value = ["/api/v1/sample/{sample_id}"],
    produces = ["application/json"],
  )
  public fun putSample(@RequestBody request: SampleUpdate?): List<Sample>?

  @GetMapping(
    value = ["/api/v1/sample/count"],
    produces = ["application/json"],
  )
  public fun getCount(): Int?

  @PutMapping(value = ["/api/v1/sample/count"])
  public fun setFlag(@RequestBody request: Boolean?)
}

package com.example.dto

import com.fasterxml.jackson.databind.PropertyNamingStrategies

import java.util.UUID

import kotlin.Long

  public val optString: String?,
  public val optLong: Long?,
  public val optDouble: Double?,
  public val optBool: Boolean?,
  public val optDate: LocalDate?,
  public val optUuid: UUID?,

  public val nullableRef: NullableInner?,
