package com.example.controller

import com.`external`.dto.NullableThing
import com.`external`.dto.PageSize
import com.`external`.dto.PlainThing
import kotlin.collections.List
import org.springframework.web.bind.`annotation`.GetMapping
import org.springframework.web.bind.`annotation`.PostMapping
import org.springframework.web.bind.`annotation`.RequestBody
import org.springframework.web.bind.`annotation`.RequestHeader
import org.springframework.web.bind.`annotation`.RequestParam

public interface Controller {
  @PostMapping(
    value = ["/a"],
    produces = ["application/json"],
  )
  public fun postNullable(@RequestBody request: NullableThing?): NullableThing?

  @GetMapping(
    value = ["/b"],
    produces = ["application/json"],
  )
  public fun getPlain(): PlainThing

  @GetMapping(
    value = ["/c"],
    produces = ["application/json"],
  )
  public fun getWithExternalDefaults(@RequestParam(required = false, name = "size", defaultValue = "20") size: PageSize, @RequestHeader(required = false, name = "X-Page-Size", defaultValue = "20") `x-PageSize`: PageSize): List<NullableThing?>
}
