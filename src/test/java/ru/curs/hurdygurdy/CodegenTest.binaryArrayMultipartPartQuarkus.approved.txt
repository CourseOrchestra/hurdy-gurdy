package com.example.controller;

import com.example.dto.UploadMeta;
import jakarta.ws.rs.Consumes;
import jakarta.ws.rs.POST;
import jakarta.ws.rs.Path;
import jakarta.ws.rs.Produces;
import java.lang.String;
import java.util.List;
import org.jboss.resteasy.reactive.RestForm;
import org.jboss.resteasy.reactive.multipart.FileUpload;

@Path("")
public interface Controller {
  @POST
  @Path("/upload")
  @Produces("application/json")
  @Consumes("multipart/form-data")
  UploadMeta upload(@RestForm("one") FileUpload one, @RestForm("files") List<FileUpload> files,
      @RestForm("attachments") List<FileUpload> attachments,
      @RestForm("alias_files") List<FileUpload> alias_files, @RestForm("tags") List<String> tags,
      @RestForm("alias_tags") List<String> alias_tags, @RestForm("meta") UploadMeta meta);
}
